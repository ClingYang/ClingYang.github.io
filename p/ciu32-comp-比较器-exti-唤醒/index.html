#include "main.h"

int main(void)
{
    /* ϵͳʱΪRCH48MHz*/
    system_clock_config();

    /* ʱʼ */
    std_delay_init();
    
    /* GPIOʼ */
    gpio_init();

    /* COMP1ʼ */
    comp1_init();

    /* NVICʼ */
    nvic_init(); 

    /* DeepStopǰʱ3sdebugʹòʱ */
    std_delayms(3000);

    /* DeepStopģʽ */
    std_pmu_enter_stop(PMU_MODE_DEEPSTOP,PMU_ENTRY_LOWPOWER_MODE_WFI);
      
    /* Ѻ󣬻ָϵͳʱ */
    system_clock_config();      

    while(1)
    {
        /* ѺLED˸ */
        std_gpio_toggle_pin(LED_GPIO_PORT,LED_PIN);
        std_delayms(300); 
    }
}

代码讲解

系统时钟初始化：配置 MCU 的主频和时钟树。不同芯片的时钟配置不同，一般由工具生成。

GPIO 配置：设置引脚为输出/输入模式，选择推挽/开漏输出，配置上拉/下拉。

中断优先级配置：NVIC 设置中断的抢占优先级和子优先级。数字越小优先级越高。
